cmake_minimum_required(VERSION 3.21)
project(PSVMSLicenseGenerator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(OpenSSL REQUIRED)
qt_add_executable(PSVMSLicenseGenerator WIN32 main.cpp)
target_link_libraries(PSVMSLicenseGenerator PRIVATE Qt6::Core Qt6::Widgets OpenSSL::Crypto)
if(MSVC)
  target_compile_options(PSVMSLicenseGenerator PRIVATE /utf-8)
endif()
